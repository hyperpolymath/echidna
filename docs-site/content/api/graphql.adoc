== GraphQL API Reference

Endpoint: `+https://localhost:8081/graphql+`

Playground: `+https://localhost:8081/playground+`

=== Schema

==== Queries

===== List Provers

[source,graphql]
----
query {
  provers {
    kind
    tier
    complexity
  }
}
----

Returns all 30 prover backends.

===== Get Proof State

[source,graphql]
----
query {
  proofState(sessionId: "session-uuid") {
    goals {
      conclusion
      hypotheses
    }
    status
    proofScript
  }
}
----

===== List Active Proofs

[source,graphql]
----
query {
  listProofs {
    sessionId
    proverKind
    status
    startedAt
  }
}
----

===== Suggest Tactics

[source,graphql]
----
query {
  suggestTactics(sessionId: "session-uuid") {
    name
    description
    confidence
  }
}
----

==== Mutations

===== Submit a Proof

[source,graphql]
----
mutation {
  submitProof(
    proverKind: COQ
    goal: "forall n : nat, n + 0 = n"
  ) {
    sessionId
    status
    goals {
      conclusion
    }
  }
}
----

===== Apply a Tactic

[source,graphql]
----
mutation {
  applyTactic(
    sessionId: "session-uuid"
    tactic: "induction n"
  ) {
    result
    goals {
      conclusion
      hypotheses
    }
  }
}
----

===== Cancel a Proof

[source,graphql]
----
mutation {
  cancelProof(sessionId: "session-uuid")
}
----
